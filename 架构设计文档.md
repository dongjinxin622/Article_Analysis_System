## 一、架构概述

### 1.1 设计目标
- **前后端分离**：Vue 前端 + FastAPI 后端，便于独立开发和部署
- **高可扩展性**：模块化设计，支持领域扩展和多数据库适配
- **云原生支持**：支持云数据库、对象存储，便于生产部署
- **高性能**：异步处理、批量优化、缓存机制

**论文向量 Collection (paper_vectors)**
```
字段定义:
- paper_id (INT64): 论文ID，主键
- user_id (INT64): 用户ID，分区键
- vector_type (VARCHAR): 向量类型 (abstract/full_text/tech_route)
- embedding (FLOAT_VECTOR): 向量数据，维度 768/1024
- metadata (JSON): 元数据 (doi, title, year)

索引:
- IVF_FLAT 索引，nlist=1024
- 支持余弦相似度 (COSINE) 搜索
```

**实体向量 Collection (entity_vectors)**
```
字段定义:
- entity_id (INT64): 实体ID，主键

- entity_type (VARCHAR): 实体类型
- entity_name (VARCHAR): 实体名称
- embedding (FLOAT_VECTOR): 向量数据
- source_sentence (VARCHAR): 来源句子

索引:
- IVF_FLAT 索引
- 支持按 entity_type 过滤检索
```

### 3.2 NER 模块

#### 3.2.1 服务设计
```python
class NERService:
    """NER 处理服务"""

    async def process_ner(
        self,
        paper_id: int,
        config_id: int,
        batch_size: int = 5
    ) -> NERResult:
        """执行 NER 处理"""
        # 1. 加载论文内容
        paper = await self._load_paper(paper_id)

        # 2. 加载领域配置
        config = await self._load_config(config_id)

        # 3. 文本分割
        sentences = await self._split_sentences(paper.content)

        # 4. 批量 LLM 处理
        entities = await self._batch_llm_process(
            sentences, config, batch_size
        )

        # 5. 存储结果
        return await self._save_result(paper_id, entities)

    async def _batch_llm_process(
        self,
        sentences: List[str],
        config: DomainConfig,
        batch_size: int
    ) -> List[Entity]:
        """批量 LLM 处理"""
        batches = self._group_sentences(sentences, batch_size)
        results = await asyncio.gather(
            *[self._call_llm(batch, config) for batch in batches]
        )
        return self._merge_entities(results)
```

```python
from pymilvus import MilvusClient
from typing import List, Dict, Any, Optional

class VectorService:
    """Milvus 向量存储与检索服务"""

    def __init__(self, config: Settings):
        self.client = MilvusClient(
            uri=f"http://{config.MILVUS_HOST}:{config.MILVUS_PORT}",
            user=config.MILVUS_USER,
            password=config.MILVUS_PASSWORD,
            db_name=config.MILVUS_DATABASE
        )
        self.embedding_dim = config.EMBEDDING_DIM

    async def insert_paper_vector(
        self,
        paper_id: int,
        user_id: int,
        vector_type: str,
        embedding: List[float],
        metadata: Dict
    ) -> bool:
        """插入论文向量"""
        data = [{
            "paper_id": paper_id,
            "user_id": user_id,
            "vector_type": vector_type,
            "embedding": embedding,
            "metadata": metadata
        }]
        return self.client.insert("paper_vectors", data)

    async def search_similar_papers(
        self,
        query_vector: List[float],
        user_id: Optional[int] = None,
        vector_type: Optional[str] = None,
        top_k: int = 10
    ) -> List[Dict[str, Any]]:
        """检索相似论文"""
        filter_expr = ""
        if user_id:
            filter_expr = f"user_id == {user_id}"
        if vector_type:
            filter_expr += f" and vector_type == '{vector_type}'"

        results = self.client.search(
            collection_name="paper_vectors",
            data=[query_vector],
            anns_field="embedding",
            limit=top_k,
            filter=filter_expr if filter_expr else None,
            output_fields=["paper_id", "vector_type", "metadata"]
        )
        return results

    async def insert_entity_vector(
        self,
        entity_id: int,
        paper_id: int,
        entity_type: str,
        entity_name: str,
        embedding: List[float],
        source_sentence: str
    ) -> bool:
        """插入实体向量"""
        data = [{
            "entity_id": entity_id,
            "paper_id": paper_id,
            "entity_type": entity_type,
            "entity_name": entity_name,
            "embedding": embedding,
            "source_sentence": source_sentence
        }]
        return self.client.insert("entity_vectors", data)

    async def search_similar_entities(
        self,
        query_vector: List[float],
        entity_type: Optional[str] = None,
        top_k: int = 20
    ) -> List[Dict[str, Any]]:
        """检索相似实体"""
        filter_expr = None
        if entity_type:
            filter_expr = f"entity_type == '{entity_type}'"

        results = self.client.search(
            collection_name="entity_vectors",
            data=[query_vector],
            anns_field="embedding",
            limit=top_k,
            filter=filter_expr,
            output_fields=["entity_id", "paper_id", "entity_name", "source_sentence"]
        )
        return results

    async def delete_paper_vectors(self, paper_id: int) -> bool:
        """删除论文相关向量"""
        self.client.delete(
            collection_name="paper_vectors",
            filter=f"paper_id == {paper_id}"
        )
        self.client.delete(
            collection_name="entity_vectors",
            filter=f"paper_id == {paper_id}"
        )
        return True
```

#### 3.3.2 Embedding 服务设计
```python
class EmbeddingService:
    """文本向量化服务"""

    def __init__(self, llm_service: LLMService):
        self.llm_service = llm_service

    async def embed_text(self, text: str) -> List[float]:
        """单文本向量化"""
        return await self.llm_service.get_embedding(text)

    async def embed_batch(self, texts: List[str]) -> List[List[float]]:
        """批量文本向量化"""
        return await self.llm_service.get_embeddings_batch(texts)

    async def embed_paper_abstract(self, paper: Paper) -> List[float]:
        """向量化论文摘要"""
        abstract = paper.abstract or ""
        return await self.embed_text(abstract)

    async def embed_paper_full_text(self, paper: Paper) -> List[float]:
        """向量化论文全文（合并关键章节）"""
        sections = paper.sections or []
        key_sections = ["abstract", "introduction", "method", "result", "conclusion"]
        text = ""
        for section in sections:
            if section.get("title", "").lower() in key_sections:
                text += section.get("content", "") + "\n"
        return await self.embed_text(text[:8000])  # 限制长度

    async def embed_entity(self, entity_name: str, context: str) -> List[float]:
        """向量化实体（带上下文）"""
        text = f"{entity_name}: {context}"
        return await self.embed_text(text)
```

```python
from abc import ABC, abstractmethod
from typing import Dict, List, Any

class DomainAnalyzerBase(ABC):
    """领域分析器基类"""

    def __init__(
        self,
        embedding_service: EmbeddingService,
        vector_service: VectorService
    ):
        self.embedding_service = embedding_service
        self.vector_service = vector_service

    @abstractmethod
    def get_entity_types(self) -> Dict[str, str]:
        """返回领域实体类型定义"""
        pass

    @abstractmethod
    def get_relation_types(self) -> Dict[str, str]:
        """返回领域关系类型定义"""
        pass

    @abstractmethod
    def get_prompt_template(self) -> str:
        """返回 NER Prompt 模板"""
        pass

    @abstractmethod
    async def build_profile(self, ner_result: NERResult) -> str:
        """构建领域特征描述"""
        pass

    @abstractmethod
    async def analyze(
        self,
        input_data: Dict[str, Any]
    ) -> Dict[str, Any]:
        """执行领域特定分析"""
        pass

    async def retrieve_similar(
        self,
        profile: str,
        domain: str,
        top_k: int = 10
    ) -> List[Dict[str, Any]]:
        """从 Milvus 检索相似论文"""
        # 1. 向量化特征描述
        vector = await self.embedding_service.embed_text(profile)

        # 2. 在 Milvus 中检索相似向量
        results = await self.vector_service.search_similar_papers(
            query_vector=vector,
            vector_type="tech_route",
            top_k=top_k
        )

        # 3. 返回相似论文列表
        return self._process_search_results(results)

    async def store_profile_vector(
        self,
        paper_id: int,
        profile: str,
        domain: str
    ) -> bool:
        """将技术路线向量存储到 Milvus"""
        vector = await self.embedding_service.embed_text(profile)
        return await self.vector_service.insert_paper_vector(
            paper_id=paper_id,
            user_id=0,  # 系统级
            vector_type="tech_route",
            embedding=vector,
            metadata={"domain": domain, "profile_text": profile}
        )
```

```python
    def list_domains(cls) -> List[str]:
```

```python
        # 从 NER 结果构建描述
        entities = ner_result.entities
        tech_params = [e for e in entities if e.type == "工艺参数"]
```

```python
from langgraph.graph import StateGraph, END

class ExtractionWorkflow:
    """批量数据提取工作流"""

    def build_graph(self) -> StateGraph:
        graph = StateGraph(ExtractionState)

        # 定义节点
        graph.add_node("upload", self.upload_node)
        graph.add_node("check_fields", self.check_fields_node)
        graph.add_node("quality_check", self.quality_check_node)
        graph.add_node("filter_data", self.filter_data_node)
        graph.add_node("config_prompt", self.config_prompt_node)
        graph.add_node("batch_process", self.batch_process_node)
        graph.add_node("hitl_confirm", self.hitl_confirm_node)
        graph.add_node("export", self.export_node)

        # 定义边
        graph.set_entry_point("upload")
        graph.add_edge("upload", "check_fields")
        graph.add_conditional_edges(
            "check_fields",
            self._check_fields_router,
            {"pass": "quality_check", "fail": END}
        )
        graph.add_edge("quality_check", "filter_data")
        graph.add_edge("filter_data", "config_prompt")
        graph.add_edge("config_prompt", "batch_process")

        graph.add_conditional_edges(
            "batch_process",
            self._success_rate_router,
            {"high": "export", "low": "hitl_confirm"}
        )
        graph.add_edge("hitl_confirm", "export")
        graph.add_edge("export", END)

        return graph.compile()
```

## 十一、Milvus 使用指南

### 11.1 Collection 初始化脚本
```python
# backend/scripts/init_milvus.py
from pymilvus import MilvusClient, DataType

def init_collections():
    """初始化 Milvus Collections"""
    client = MilvusClient(uri=settings.MILVUS_URI)

    # 创建论文向量 Collection
    if not client.has_collection("paper_vectors"):
        client.create_collection(
            collection_name="paper_vectors",
            dimension=settings.EMBEDDING_DIM,
            id_type="Int64",
            auto_id=False,
            fields=[
                {"name": "paper_id", "dtype": DataType.INT64, "is_primary": True},
                {"name": "user_id", "dtype": DataType.INT64},
                {"name": "vector_type", "dtype": DataType.VARCHAR, "max_length": 50},
                {"name": "embedding", "dtype": DataType.FLOAT_VECTOR, "dim": settings.EMBEDDING_DIM},
                {"name": "metadata", "dtype": DataType.JSON}
            ]
        )

    # 创建实体向量 Collection
    if not client.has_collection("entity_vectors"):
        client.create_collection(
            collection_name="entity_vectors",
            dimension=settings.EMBEDDING_DIM,
            fields=[
                {"name": "entity_id", "dtype": DataType.INT64, "is_primary": True},
                {"name": "paper_id", "dtype": DataType.INT64},
                {"name": "entity_type", "dtype": DataType.VARCHAR, "max_length": 100},
                {"name": "entity_name", "dtype": DataType.VARCHAR, "max_length": 200},
                {"name": "embedding", "dtype": DataType.FLOAT_VECTOR, "dim": settings.EMBEDDING_DIM},
                {"name": "source_sentence", "dtype": DataType.VARCHAR, "max_length": 1000}
            ]
        )

    print("Milvus Collections initialized successfully.")
```
